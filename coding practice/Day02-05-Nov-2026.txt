##Day02 — Services & Tags

#Ansible script — Install & Enable Nginx
- name: Install and start Nginx on Pathnex
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present

    - name: Enable nginx
      service:
        name: nginx
        state: started
        enabled: yes

### Kubernetes — Deployment with 2 Replicas
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mypathnex-deployment
spec:
  replicas: 11
  selector:
    matchLabels:
      app: mypathnex-app
  template:
    metadata:
      labels:
        app: mypathnex-app
    spec:
      containers:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
