###Day03 Ansible, Terraform & Kubernetes Basics


###Ansible playbook — Install Apache and Start its Service
- name: Install and Start Apache
  hosts: all
  become: yes
  tasks:
    - name: Install apache
      yum:
        name: httpd
        state: present

    - name: Start apache
      service:
        name: httpd
        state: started
        enabled: yes

###Kubernetes — Create a ReplicaSet with 11 Replicas
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: shoib-replicaset
spec:
  replicas: 11
  selector:
    matchLabels:
      app: shoib-app
  template:
    metadata:
      labels:
        app: shoib-app
    spec:
      containers:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
